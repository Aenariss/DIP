\label{json-appendix}

\setcounter{listing}{0}
\begin{listing}[!ht]
    \begin{minted}[frame=none, breaklines]{json}
    {
        "pubmatic.com": {
            "ads": {
                "A": ["23.75.64.240"],
                "CNAME": [
                    "pubmatic.edgekey.net",
                    "e6603.g.akamaiedge.net"
                ]
            }
        },
        "edgekey.net": {
            "pubmatic": {
                "A": [],
                "CNAME": ["e6603.g.akamaiedge.net"]
            }
        },
        "akamaiedge.net": {
            "e6603.g": {
                "A": ["23.75.64.240"],
                "CNAME": []
            }
        }
    }
    \end{minted}
    \caption{Example of a logged DNS response for \texttt{ads.pubmatic.com} with both \textit{A} and \textit{CNAME} records. For each logged \textit{CNAME} for \texttt{ads.pubmatic.com}, another separate record is made, containing only the record for the following \textit{CNAME} in the original list. The record for the last \textit{CNAME} in the original list has the same \textit{A} response as the original request for \texttt{ads.pubmatic.com}.}
    \label{stored-dns-figure}
\end{listing}
